  console.log("Server is running on http://localhost:3000");

// 3. Our new Login Endpoint
app.post("/api/login", function(request, response) {
  const loginAttempt = request.body; // Contains the email and password they typed
  
  // 4. Search the Filing Cabinet
  // We look for a user whose email AND password match the login attempt
  const foundUser = usersDatabase.find(function(user) {
    return user.emailAddress === loginAttempt.email && user.userPassword === loginAttempt.password;
  });

  // 5. Did we find them?
  if (foundUser) {
    // Yes! Send success and their name back to React
    response.json({
      status: "Success",
      message: "Welcome back!",
      name: foundUser.fullName
    });
  } else {
    // No! Send an error
    response.status(401).json({
      status: "Error",
      message: "Invalid email or password."
    });
  }
});
